# 2. Реализация программы
~~~
#define _CRT_SECURE_NO_WARNINGS
#include <stdlib.h>
#include <stdio.h>
#include <locale.h>
#include <math.h>

int main() {
    setlocale(LC_CTYPE, "RUS");
    int n;
    double product = 1.0;

    printf("Введите целое число n (n > 0): ");
    if (scanf("%d", &n) != 1 || n <= 0) {
        printf("Ошибка! n должно быть больше 0.\n");
        return 1;
    }

    for (int i = 1; i <= n; i++) {
        product *= (1.0 + 1.0 / i);
    }

    printf("Результат вычисления для n = %d: %.2f\n", n, product);

    return 0;
}
~~~
